cmake_minimum_required(VERSION 3.24)
project(OpenRendNative VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(FFMPEG_ROOT "${CMAKE_SOURCE_DIR}/.deps/ffmpeg" CACHE PATH "FFmpeg shared SDK")
if(NOT WIN32)
  message(FATAL_ERROR "OpenRend's D3D11 engine requires Windows")
endif()
add_library(OpenRend.Native SHARED src/OpenRend.Native/media.cpp src/OpenRend.Native/renderer.cpp src/OpenRend.Native/compositor.cpp)
target_compile_definitions(OpenRend.Native PRIVATE OPENREND_NATIVE_EXPORTS WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(OpenRend.Native PUBLIC src/OpenRend.Native)
target_include_directories(OpenRend.Native SYSTEM PRIVATE "${FFMPEG_ROOT}/include")
foreach(component avformat avcodec avutil swscale swresample)
  find_library(FFMPEG_${component} NAMES ${component} PATHS "${FFMPEG_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
  target_link_libraries(OpenRend.Native PRIVATE "${FFMPEG_${component}}")
endforeach()
target_link_libraries(OpenRend.Native PRIVATE d3d11 dxgi d3dcompiler)
if(MSVC)
  target_compile_options(OpenRend.Native PRIVATE /W4 /EHsc /utf-8)
endif()
file(GLOB FFMPEG_RUNTIME "${FFMPEG_ROOT}/bin/*.dll")
add_custom_command(TARGET OpenRend.Native POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different ${FFMPEG_RUNTIME} "$<TARGET_FILE_DIR:OpenRend.Native>" COMMAND_EXPAND_LISTS)
option(OPENREND_BUILD_TESTS "Build native integration tests" ON)
if(OPENREND_BUILD_TESTS)
  include(FetchContent)
  set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
  FetchContent_Declare(googletest URL https://github.com/google/googletest/archive/refs/tags/v1.17.0.zip DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(googletest)
  enable_testing()
  add_executable(OpenRend.Native.Tests tests/native/media_tests.cpp)
  target_compile_definitions(OpenRend.Native.Tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(OpenRend.Native.Tests PRIVATE OpenRend.Native GTest::gtest_main user32)
  target_compile_definitions(OpenRend.Native.Tests PRIVATE OR_TEST_MEDIA="${CMAKE_SOURCE_DIR}/tests/native/fixture.mp4" OR_TEST_AUDIO="${CMAKE_SOURCE_DIR}/tests/native/fixture.wav")
  add_test(NAME native_integration COMMAND OpenRend.Native.Tests)
endif()
